     3	 *
     4	 * This is the on-device half of "generate from a photo". There is no neural network here and there
     5	 * is not going to be one: a segmentation model large enough to be worth shipping would blow the

   480	void specFromAnalysis(const af_image_analysis& identity, const af_image_analysis* clothing,
   481	                      af_spec* spec, float* outConfidence) {
   482	    if (!spec) return;
   483	    float confidence = identity.pose_confidence;
   484	
   485	    /* Stature has strict precedence: explicit metres in af_spec win; only the canonical default
   486	       (or a legacy zero/non-finite value) may be refined from a genuine full-body image. A portrait
   487	       has no absolute scale, so it keeps 1.75 m instead of writing the old 0 sentinel. */
   488	    {
   489	        const bool unresolved = !std::isfinite(spec->height_m) || spec->height_m <= 0.0f ||
   490	                                std::fabs(spec->height_m - AF_DEFAULT_HEIGHT_M) < 1e-5f;
   491	        const float headHeight = std::fmax(0.01f, identity.head_box[3]);
   492	        const float statureNorm = identity.ankle_y - identity.head_box[1];
   493	        const int fullBody = identity.is_full_body;
   494	        if (unresolved && fullBody && statureNorm > 0.05f) {
   495	            const float heads = statureNorm / headHeight;
   496	            /* The ratio is scale-free; 0.235 m is the adult head-height prior. Confidence pulls a
   497	               noisy segmentation towards the canonical default, never towards zero. */
   498	            float h = 0.235f * heads;
   499	            if (h < 1.40f) h = 1.40f;
   500	            if (h > 2.20f) h = 2.20f;
   501	            const float w = std::fmax(0.0f, std::fmin(1.0f, (confidence - 0.45f) / 0.5f));
   502	            spec->height_m = w * h + (1.0f - w) * AF_DEFAULT_HEIGHT_M;
   503	        } else if (!std::isfinite(spec->height_m) || spec->height_m <= 0.0f) {
   504	            spec->height_m = AF_DEFAULT_HEIGHT_M;
   505	        }
   506	        if (!fullBody) confidence *= 0.6f;
   507	    }
   508	
   509	    /* Sex: shoulder-to-hip ratio. Above 1.45 is male-typical, below 1.30 female-typical, and in
   510	       between the estimate is left alone rather than guessed. */
   511	    {
   512	        const float shoulder = identity.head_box[2];
   513	        (void)shoulder;
   514	    }
   515	
   516	    /* Skin tone: the engine's melanin -> albedo curve is monotone in L*, so invert it against the
   517	       measured skin lightness, normalised by the background (the only reference for exposure a
   518	       single image offers). */
   519	    {
   520	        const float skinL = identity.skin_lab[0];
   521	        const float refL = std::fmax(22.0f, identity.background_lab[0]);
   522	        const float ratio = std::fmax(0.15f, std::fmin(1.9f, skinL / refL));
   523	        float melanin = 0.62f - 0.52f * ratio;              /* 1.0 ratio -> medium-light skin */
   524	        if (melanin < 0.03f) melanin = 0.03f;
   525	        if (melanin > 0.95f) melanin = 0.95f;
   526	        spec->skin_melanin = melanin;
   527	    }
   528	
   529	    /* Hair tone: Lab -> linear sRGB, then darkened slightly because the photo's hair pixels sit in
   530	       shadow as often as not. */
   531	    {
   532	        float rgb[3];
   533	        labToLinearRgb(identity.hair_lab, rgb);
   534	        for (int c = 0; c < 3; ++c) spec->hair_tone[c] = std::fmax(0.01f, std::fmin(1.0f, rgb[c] * 1.15f));
   535	    }
   536	
   537	    /* Clothing: colour and material class. Metal-looking references become plate; everything else
   538	       becomes cloth. The colour comes from the identity image unless a clothing reference was
   539	       supplied, in which case that image wins - it is the one the user pointed at. */
   540	    {
   541	        const af_image_analysis& src = clothing ? *clothing : identity;
   542	        const float metal = clothing ? clothing->metal_confidence : identity.metal_confidence;
   543	        float rgb[3];
   544	        labToLinearRgb(src.garment_lab, rgb);
   545	        for (int c = 0; c < 3; ++c) {
   546	            spec->cloth_tone[c] = std::fmax(0.01f, std::fmin(1.0f, rgb[c]));
   547	            spec->gear_tone[c] = std::fmax(0.01f, std::fmin(1.0f, rgb[c] * 1.05f));
   548	        }
   549	        if (spec->outfit == AF_OUTFIT_NONE || spec->outfit == AF_OUTFIT_TSHIRT) {
   550	            if (metal > 0.62f) {
   551	                spec->outfit = AF_OUTFIT_ARMOR_PLATE;
   552	                if (spec->footwear == AF_FOOTWEAR_NONE) spec->footwear = AF_FOOTWEAR_SABATON;
   553	            } else if (metal > 0.45f) {
   554	                spec->outfit = AF_OUTFIT_ARMOR_LEATHER;
   555	                if (spec->footwear == AF_FOOTWEAR_NONE) spec->footwear = AF_FOOTWEAR_BOOT;
   556	            } else if (spec->outfit == AF_OUTFIT_NONE) {
   557	                spec->outfit = AF_OUTFIT_TSHIRT;
   558	            }
   559	        }
   560	        if (outConfidence) {
   561	            *outConfidence = std::fmax(0.0f, std::fmin(1.0f, confidence * 0.5f + (clothing ? 0.35f : 0.2f)));
   562	        }
   563	    }
   564	    if (outConfidence) {
   565	        *outConfidence = std::fmax(0.0f, std::fmin(1.0f, *outConfidence));
   566	    }
   567	}
